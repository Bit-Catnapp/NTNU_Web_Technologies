///Main Function
function formatTxt(str, callback) {
    //Check if the input is a string
    if (typeof str !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        return callback(str);
    }
}

///Callbacks
function toShouting(str) {
    //Check if the input is a string
    if (typeof str !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        return str.toUpperCase();
    }
}
function toSecret(str) {
    //Check if the input is a string
    if (typeof str !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        return str.replace(/[aeiou]/gi, '*');
    }
}

//Helper
function showResultTxt(inputId, callback) {
    //Retrieve id from input
    const textInput = document.getElementById(inputId);

    //Call the formatTxt function with the input and callback
    const result = formatTxt(textInput.value, callback);
    textInput.value = result;
    alert(result);

    //Reset the input field to empty after displaying the result
    document.getElementById(inputId).value = '';
}

//main function 
function randomNum() {
    return Math.floor(Math.random() * 10) + 1;
};

//callbacks
function double(num) {
    if (typeof num !== 'number') {
        throw new Error('Input must be a number');
    }
    else {
        return num * 2;
    }
}

function even_or_odd(num) {
    if (typeof num !== 'number') {
        throw new Error('Input must be a number');
    }
    else {
        //using ternary operator and modulo flag check if value is divisible by 2 or not
        //If divisible by 2 then it is even else odd
        return num % 2 === 0 ? 'even' : 'odd';
    }
}

//Helper
function showResultNum(callback) {
    //Retrieve id from input
    const randomNumber = randomNum();
    const resultNum = callback(randomNumber);

    alert("Number was : " + randomNumber + " and result is : " + resultNum);
}

//Main function
function sendCampaign(campaignName, callback) {
    // Local variables 
    let seconds = 3;
    let countdown;

    // Display the status of the campaign
    document.getElementById("status").innerHTML =
        "Initiating transmission sequence for: " + campaignName + "<br>";

    // Countdown timer for sending the campaign
    document.getElementById("timer").innerHTML =
        "Sending in " + seconds + "...";

    // Start the countdown timer and call the countdownCallback function every second
        countdown = setInterval(() => {
            // Call the countdownCallback function and update the seconds variable
            seconds = countdownCallback(campaignName, callback, seconds, countdown);
        }, 1000);
}

//Callbacks
function countdownCallback(campaignName, callback, seconds, countdown) {
    // Decrement the seconds and update the timer display
    seconds--;

    // If there are still seconds left, update the timer display
    if (seconds > 0) {
        document.getElementById("timer").innerHTML =
            "\nSending in " + seconds + "...";
    }
    else {
        // If the countdown has reached zero, clear the interval and call the callback function
        clearInterval(countdown);
        document.getElementById("timer").innerHTML = "";
        callback(campaignName);

        // Clear the status and timer display after the campaign is sent
        document.getElementById("status").innerHTML = "";
        document.getElementById("timer").innerHTML = "";
    }

    // Return the remaining seconds for testing purposes
    return seconds;
}

function generateReport(campaignName) {
    // Display the result after the campaign is sent
    document.getElementById('result').innerHTML = 
        "Performance analytics report compiled and updated on dashboard.<br>" +
        "Emails delivered successfully to: " + campaignName;

    // Clear the result after 3 seconds
    // () => means: call this function every x-seconds
    setTimeout(() => {
        document.getElementById('result').innerHTML = "";
    }, 3000);           
}

//Show result
function showResultCampaign(campaignNameID) {
    // Get the campaign name from the input field
    const campaignName = document.getElementById(campaignNameID).value;
    sendCampaign(campaignName, generateReport);

    // Clear the input field after sending the campaign
    document.getElementById(campaignNameID).value = "";
}

//Main function
function calculateFactorial(number, callback) {
    let seconds = 3;
    let countdown;

    document.getElementById("statusCalc").innerHTML =
        "Initializing factorial calculation for: " + number + "<br>";

    // Countdown timer for sending the campaign
    document.getElementById("timerCalc").innerHTML =
        "Sending in " + seconds + "...";
    
    //Calculate factorial
    const fact_num = factorial(number);

    // Start the countdown timer and call the countdownCallback function every second
    countdown = setInterval(() => {
       
        // Call the countdownCallback function and update the seconds variable
        seconds = countdownCallback(fact_num, callback, seconds, countdown);
    }, 1000);
}

//Callbacks
function countdownCallback(number, callback, seconds, countdown) {
    // Decrement the seconds and update the timer display
    seconds--;

    // If there are still seconds left, update the timer display
    if (seconds > 0) {
        document.getElementById("timerCalc").innerHTML =
            "\nSending in " + seconds + "...";
    }
    else {
        // If the countdown has reached zero, clear the interval and call the callback function
        clearInterval(countdown);
        document.getElementById("timerCalc").innerHTML = "";
        callback(number);

        // Clear the status and timer display after the campaign is sent
        document.getElementById("statusCalc").innerHTML = "";
        document.getElementById("timerCalc").innerHTML = "";
    }

    // Return the remaining seconds for testing purposes
    return seconds;
}

function displayResult(number) {
    // Display the result after the campaign is sent
    document.getElementById('resultCalc').innerHTML = 
        "Calculation complete.<br>" +
        "The final computed result is: " + number;

    // Clear the result after 3 seconds
    setTimeout(() => {
        document.getElementById('resultCalc').innerHTML = "";
    }, 1500);           
}

function factorial(n) {
    // Handle negative numbers
    if (n < 0) {
        return "Error: Factorial is not defined for negative numbers."; 
    }

    // Base case: factorial of 0 is 1
    if (n === 0) {
        return 1;
    }

    // Recursive case: n! = n * (n-1)!
    return n * factorial(n - 1);
}

//Show result
function showResultCalculation(inputID) {
    // Get the number from the input field
    const number = document.getElementById(inputID).value;
    calculateFactorial(number, displayResult);

    // Clear the input field after calculating the factorial    
    document.getElementById(inputID).value = "";
}
