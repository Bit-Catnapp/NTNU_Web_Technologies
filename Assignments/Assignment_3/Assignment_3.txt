///Callbacks
function toShouting(str) {
    //Check if the input is a string
    if (typeof str !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        return str.toUpperCase();
    }
}
function toSecret(str) {
    //Check if the input is a string
    if (typeof str !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        return str.replace(/[aeiou]/gi, '*');
    }
}

///Main Function
function formatTxt(inputId, callback) {

    // Retrieve string from inputID
    const textInput = document.getElementById(inputId).value;

    //Check if the input is a string
    if (typeof textInput !== 'string') {
        throw new Error('Input must be a string');
    }
    else {
        //Call the formatTxt function with the input and callback
        const result = callback(textInput);
        textInput.value = result;
        alert(result);

        //Reset the input field to empty after displaying the result
        document.getElementById(inputId).value = '';
    }
}

function double(num) {
    if (typeof num !== 'number') {
        throw new Error('Input must be a number');
    }
    else {
        return num * 2;
    }
}

function even_or_odd(num) {
    if (typeof num !== 'number') {
        throw new Error('Input must be a number');
    }
    else {
        //using ternary operator and modulo flag check if value is divisible by 2 or not
        //If divisible by 2 then it is even else odd
        return num % 2 === 0 ? 'even' : 'odd';
    }
}

//Callbacks
function countdownCallback(campaignName, callback, seconds, countdown) {
    // Decrement the seconds and update the timer display
    seconds--;

    // If there are still seconds left, update the timer display
    if (seconds > 0) {
        document.getElementById("timer").innerHTML =
            `Sending in ${seconds}...`;
    }
    else {
        // If the countdown has reached zero, clear the interval and call the callback function
        clearInterval(countdown);
        document.getElementById("timer").innerHTML = "";
        callback(campaignName);

        // Clear the status and timer display after the campaign is sent
        document.getElementById("status").innerHTML = "";
        document.getElementById("timer").innerHTML = "";
    }
}

function generateReport(campaignName) {
    // Display the result after the campaign is sent
    document.getElementById('result').innerHTML =
        `Performance analytics report compiled and updated on dashboard.<br>` +
        `Emails delivered successfully to: ${campaignName}`;

    // Clear the result after 3 seconds
    setTimeout(() => {
        document.getElementById('result').innerHTML = "";
    }, 3000);
}

//Main function
function sendCampaign(campaignNameID) {
    // Local variables 
    const campaignName = document.getElementById(campaignNameID).value;
    let seconds = 3;
    let countdown;

    // Display the status of the campaign
    document.getElementById("status").innerHTML =
        `Initiating transmission sequence for: ${campaignName}<br>`;

    // Countdown timer for sending the campaign
    document.getElementById("timer").innerHTML =
        `Sending in ${seconds}...`;

    // Start the countdown timer and call the countdownCallback function every second
    countdown = setInterval(() => {
        // Call the countdownCallback function and update the seconds variable
        seconds = countdownCallback(campaignName, generateReport, seconds, countdown);
    }, 1000);

    // Clear the input field after sending the campaign
    document.getElementById(campaignNameID).value = "";
}

function factorial(n) {
    // Handle negative numbers
    if (n < 0) return "Error: Factorial is not defined for negative numbers.";
    // Base case: factorial of 0 is 1
    if (n === 0) return 1;

    // Recursive case: n! = n * (n-1)!
    return n * factorial(n - 1);
}

//Main function
function calculateFactorial(inputID) {
    // Get the number from the input field
    const number = document.getElementById(inputID).value;
    let seconds = 1.5;
    let countdown;

    document.getElementById("statusCalc").innerHTML =
        `Initializing factorial calculation for: ${number}<br>`;

    // Countdown timer for sending the campaign
    document.getElementById("timerCalc").innerHTML =
        `Result ready in ${seconds}...`;

    //Calculate factorial
    const fact_num = factorial(number);

    // Start the countdown timer and call the countdownCallback function every second
    countdown = setInterval(() => {

        // Call the countdownCallback function and update the seconds variable
        seconds = countdownCallbackCalc(fact_num, displayResult, seconds, countdown);
    }, 1000);

    // Clear the input field after calculating the factorial    
    document.getElementById(inputID).value = "";
}

//showResult function
function PlayDiceRoll() {
    // Local variables
    let displayID = document.getElementById("diceRolls");
    const localRandomNumber = Math.floor(Math.random() * 6) + 1;

    //Simulate dice roll
    const diceRoll = new Promise((resolve, reject) => {

        //Check if the roll is high or low
        if (localRandomNumber > 3) {
            resolve(`Rolled ${localRandomNumber}: High roll! You win.`);
        }
        else {
            reject(`Rolled ${localRandomNumber}: Low roll! You lose.`);
        }
    });

    //Handle the promise and log the result to the console
    diceRoll
        .then((result) => displayID.innerHTML = `Dice Roll: ${localRandomNumber}<br>${result}`)
        .catch((error) => displayID.innerHTML = `Dice Roll: ${localRandomNumber}<br>${error}`)
        .finally(() => {

            // Display message for timer
            const resetMessage = displayID.innerHTML;
            let timer = 3;

            //Display Interval for resetting
            const countdown = setInterval(() => {

                //Updating resetMessage
                displayID.innerHTML =
                    `${resetMessage}<br>Dice Roll Complete<br>Resetting in ${timer}...`;

                //de-incrementing timer
                timer--;

                //if timer is less than 0, clear timer and reset
                if (timer < 0) {
                    clearInterval(countdown);
                    displayID.innerHTML = "";
                }

            }, 1000);
        });
};

// showResult function
function playShopping() {
    // Shoppingcart object
    const shoppingCart = {
        coupon: document.getElementById("coupon").value,
        expired: document.getElementById("expired").checked,
        price: document.getElementById("original_price").value
    };

    // Promise
    const validateCoupon = new Promise((resolve, reject) => {

        // Promise chaining
        if (shoppingCart.coupon < 0) {
            reject("Coupon value illegal");

        }
        else if (shoppingCart.expired) {
            reject("Expired Code");

        }
        else {
            resolve(shoppingCart);
        }
    });

    //Handle the promise and log the result to the console
    validateCoupon
        // Output new price if promise was successfull
        .then((shoppingCart) => {
            const newPrice = shoppingCart.price * (1 - shoppingCart.coupon / 100);
            const diff = shoppingCart.price - newPrice;

            //Update the element variable and display the result in the HTML
            document.getElementById("discount").innerHTML =
                `Success!<br>
                 Coupon discount was: ${shoppingCart.coupon}<br>
                 Not expired: ${shoppingCart.expired}<br>
                 Original price was: ${shoppingCart.price}<br> 
                 You saved: ${diff}<br> 
                 New price is: ${newPrice}<br>
                 Session sync complete...`;
        })
        .catch((error) => {
            document.getElementById("discount").innerHTML = error
        })
        .finally(() => {
            //Display Timer
            setTimeout(() => {
                document.getElementById("shoppingCartForm").reset();
                document.getElementById("discount").innerHTML = "";
            }, 3000); // 1000 ms = 1 second
        });
};
